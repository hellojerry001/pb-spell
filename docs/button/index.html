UI.btn("刷新", { act: "refresh-all" })
// → <button class="btn" data-act="refresh-all">刷新</button>

// → <button class="btn is-ghost push-end" data-act="delete-open" data-id="p1">删除</button>

工具条里的分工：左 lead、右主力
/**
 * Button —— 按钮
 * 类名与 PortButler 生成的 HTML 完全一致（class="btn is-ghost" 等）。
 * `pb-show-hover` 是文档专用类，把各变体的 hover 效果静态呈现出来。
 */
export function Demo() {
  return (
    <div className="pb-stack">
      <div>
        <span className="pb-spec">四种变体 · 静置</span>
        <div className="pb-row">
          <button className="btn">刷新</button>
          <button className="btn is-ghost">手动添加</button>
          <button className="btn is-primary">＋ 从模板新建</button>
          <button className="btn is-danger">删除</button>
        </div>
      </div>

      <div>
        <span className="pb-spec">四种变体 · hover</span>
        <div className="pb-row">
          <button className="btn pb-show-hover">刷新</button>
          <button className="btn is-ghost pb-show-hover">手动添加</button>
          <button className="btn is-primary pb-show-hover">＋ 从模板新建</button>
          <button className="btn is-danger pb-show-hover">删除</button>
        </div>
      </div>

      <div>
        <span className="pb-spec">禁用 · 计数角标</span>
        <div className="pb-row">
          <button className="btn" disabled>
            刷新
          </button>
          <button className="btn is-ghost">
            发布记录
            <span className="btn-count">3</span>
          </button>
        </div>
      </div>

      <div>
        <span className="pb-spec">工具条里的分工：左 lead、右主力</span>
        <div className="toolbar" style={{ marginBottom: 0 }}>
          <button className="btn is-ghost toolbar-lead">
            发布记录
            <span className="btn-count">3</span>
          </button>
          <button className="btn is-ghost">手动添加</button>
          <button className="btn is-primary">＋ 从模板新建</button>
        </div>
      </div>
    </div>
  );
}

变体

类名静置hover用在哪
.btn--c-surface-3 底--c-surface-6常规动作、弹窗里的主提交（保存 / 创建并启动）
.btn.is-ghost透明 + --c-text-2 字--c-surface-3 底 + --c-text 字次要动作、卡片里的一排操作按钮
.btn.is-primary--c-primary-bg 底 + --c-primary-fg 字 + 600 字重--c-primary-bg-hover一屏一个的主操作
.btn.is-danger与默认完全相同--c-danger-bg 底 + 反白字删除类动作

危险按钮静置时与默认按钮长得一样，只有 hover 才染红——这是刻意的： 危险操作不该在视觉上抢注意力，但一旦鼠标真的移上去，必须给出明确警告。

主操作按钮的底色与前景是成对声明的（--c-primary-bg / --c-primary-fg）。 深色下主按钮只是「比普通按钮亮一阶」，浅色下则直接是近黑实底 + 反白字—— 浅灰底上再叠浅灰等于没做，必须换个强对比的做法。

修饰类

类名效果说明
.push-endmargin-left: auto把元素推到行尾。用在卡片里让「删除」独占最右
.toolbar-leadmargin-right: auto工具条里把左侧元素顶开，其余保持右对齐
.btn-count左侧 5px 外边距 + --c-text-3按钮内的计数角标，如「发布记录 3」
.btn[disabled]opacity: .45 + cursor: default禁用态

相关令牌

.btn {
  padding: var(--px-5) var(--px-12);
  border-radius: var(--r-7);
  font-size: var(--fs-12);
  white-space: nowrap;   /* 卡片网格换列时不把文字竖着拆开 */
}

white-space: nowrap 是踩坑后加的：3 列网格下「＋ 从模板新建」曾被拆成竖排。

注意

  • 按钮不设 font-family。原生控件的字体族在 base.css 里刻意不覆写——原实现依赖 UA 默认， 改了会变字形。要复现这一点，文档站的预览作用域里把 font 重置回了 UA （Tailwind 的 preflight 会写 font: inherit，两者不一致，见 preview-scope.css）。
  • 所有按钮的事件都写 data-act 交给文档级委托层，不在元素上绑 onclick。
