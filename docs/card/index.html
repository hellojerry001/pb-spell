/**
 * Card —— 项目卡片
 * 结构来自 projects.js 的 renderProjects()：card-head（dot + [card-icon] + name
 * + tag + port-chip）/ card-meta / card-ops。
 *
 * 两类项目的差异全在操作集合与标记上：Web 有「浏览器 / 发布」与端口 chip；
 * Mac 有「开发预览 / 打包 DMG」、额外一个 Mac tag、端口为 0 时不渲染
 * port-chip，且卡片头多一个图标缩略图（.card-icon）。
 */

const OPS_WEB_LIVE = ["停止", "浏览器", "发布", "编辑", "删除"];
const OPS_WEB_IDLE = ["启动", "浏览器", "发布", "编辑", "删除"];
const OPS_MAC_LIVE = ["停止", "打包 DMG", "编辑", "删除"];
const OPS_MAC_IDLE = ["开发预览", "打包 DMG", "编辑", "删除"];

function Card({
  name,
  scaffold,
  port,
  path,
  command,
  live,
  mac,
}: {
  name: string;
  scaffold?: string;
  port: number;
  path: string;
  command: string;
  live?: boolean;
  mac?: boolean;
}) {
  const ops = mac
    ? live
      ? OPS_MAC_LIVE
      : OPS_MAC_IDLE
    : live
      ? OPS_WEB_LIVE
      : OPS_WEB_IDLE;

  return (
    <div className="card">
      <div className="card-head">
        <span className={live ? "dot is-on" : "dot"}></span>
        {/* Mac 项目才有图标缩略图；没读到图标时留 is-empty 占位 */}
        {mac ? <span className="card-icon is-empty"></span> : null}
        <span className="card-name">{name}</span>
        {mac ? <span className="tag">Mac</span> : null}
        {scaffold ? <span className="tag">{scaffold}</span> : null}
        {port ? <span className="port-chip">:{port}</span> : null}
      </div>
      <div className="card-meta" title={path}>
        {path} · {command}
      </div>
      <div className="card-ops">
        {ops.map((label, i) => {
          const isDanger = label === "删除";
          const last = i === ops.length - 1;
          return (
            <button
              // 删除按钮在真实应用里带 push-end 靠右
              key={label}
              className={"btn is-ghost" + (isDanger && last ? " push-end" : "")}
            >
              {label}
            </button>
          );
        })}
      </div>
    </div>
  );
}

// Mac 项目才挂图标缩略图；没读到图标时留 is-empty 占位
const icon = mac
  ? '<span class="card-icon' + (macIcons.get(p.id) ? "" : " is-empty") +
    '"' + (macIcons.get(p.id)
      ? ' style="background-image:url(data:image/png;base64,' + macIcons.get(p.id) + ')"'
      : "") + "></span>"
  : "";

return '<div class="card">' +
    '<div class="card-head">' +
      UI.dot(live) +
      icon +
      '<span class="card-name">' + esc(p.name) + "</span>" +
      (mac ? '<span class="tag">Mac</span>' : "") +
      (p.scaffold ? '<span class="tag">' + esc(p.scaffold) + "</span>" : "") +
      (p.port ? '<span class="port-chip">:' + esc(p.port) + "</span>" : "") +
    "</div>" +
    '<div class="card-meta" title="' + esc(p.path) + '">' +
      esc(p.path) + " · " + esc(p.command) +
    "</div>" +
    '<div class="card-ops">' + ops + "</div>" +
  "</div>";
类名作用
.card--c-surface-1 底 + --hairline 描边 + --r-12 + 14px 16px 内边距；min-width: 0 允许被网格压缩
.card-head卡片头一行：flex + gap: 8px，承载 dot / card-icon / name / tag / port-chip
.card-iconMac 项目的启动图标缩略图：--rail-btn 见方 + --r-10 + --c-surface-3 底 + background-size: cover；flex: none。is-empty 时底色本身就是占位（::after 撑住盒模型），避免布局跳动
.card-name--fs-14 + 600 字重，超长单行省略号（overflow:hidden + text-overflow:ellipsis + white-space:nowrap）
.card-meta路径 + 启动命令，--fs-11 + --c-text-3，同样单行省略；title 属性放完整路径
.card-ops操作按钮行：flex + wrap + gap: 4px + margin-top: 10px

图标缩略图的数据流

图标不随项目列表一起下发——列表接口不该背着几百 KB 的 base64。改成先渲染再补图：

async function loadMacIcons() {
  const macs = projects.filter(p => kindOf(p) === "mac");
  if (!macs.length) return;
  await Promise.all(macs.map(async p => {
    try {
      const d = await invoke("project_icon", { id: p.id });   // 裸 base64
      if (d) macIcons.set(p.id, d);
    } catch { /* 没图标就留占位 */ }
  }));
  renderProjects();   // 拿到图后重渲染一次
}

refreshAll() 末尾调它（不 await，列表先出来）。在打包弹窗里换完图标后 macIcons.delete(id) 再重拉，缩略图立即跟上。project_icon 返回的是裸 base64，data: 前缀由前端拼。

卡片网格

.card-grid {
  display: grid;
  grid-template-columns: repeat(auto-fill, minmax(var(--grid-min), 1fr));
  gap: var(--gap-row);
}

--grid-min 是 320px（早期是 290px）。改大的原因是 3 列布局下， 290px 宽度里「＋ 从模板新建」这个较长的按钮文字被拆成竖排；同时给 .btn 加了 white-space: nowrap 双保险。

card-ops 上的 flex-wrap: wrap 也是同一批修复：按钮在窄卡片里宁可换行，也不要被压扁。

操作按钮的构成

const ops = [
  live
    ? UI.btn("停止", { variant: "ghost", act: "op", data: { op: "stop", id: p.id } })
    : UI.btn(mac ? "开发预览" : "启动", { act: "op", data: { op: "start", id: p.id } }),
  mac
    ? UI.btn("打包 DMG", { variant: "ghost", act: "build-open",  data: { id: p.id } })
    : UI.btn("浏览器",   { variant: "ghost", act: "browser",     data: { port: p.port } }),
  mac ? "" : UI.btn("发布", { variant: "ghost", act: "publish-open", data: { id: p.id } }),
  UI.btn("编辑", { variant: "ghost", act: "project-edit", data: { id: p.id } }),
  UI.btn("删除", { variant: "ghost", act: "delete-open",  data: { id: p.id }, push: true }),
].join("");

规律：运行中才用实底按钮（「停止」），其余全是 ghost；「删除」加 push: true 靠 margin-left:auto 独占最右，与其余操作拉开距离。

改图标与改应用名不在卡片上单独开口子——它们属于「打包前要定的东西」， 统一收进 打包 DMG 弹窗的配置态（见下方「图标与应用名的编辑入口」）。

两类项目的卡片差异

卡片外形只有一套，差异全在操作集合和几个小标记上：

Web 项目Mac 项目
卡片头dot + name + tag + port-chip多一个图标缩略图（.card-icon）与 Mac tag
启动按钮启动开发预览（起的是原生窗口）
第二个操作浏览器 → localhost:<port>打包 DMG → 打开打包弹窗（配置态 → 打包态）
第三个操作发布 → Cloudflare无（.dmg 不是静态站点）
来源标记脚手架 key（如 pc）额外多一个 Mac tag
端口标签必有端口为 0 时整个不渲染

Mac tag 与脚手架 tag 共用 .tag，靠 flex: none 保证长名称被省略时它不被压掉。

图标与应用名的编辑入口

改图标、改应用名都不是独立功能，而是打包前的准备动作——所以它们住在打包弹窗里， 卡片上不再有单独的入口。弹窗因此分两态：

态内容底部按钮
配置态应用图标（可点换）、应用名称、应用标识（只读）取消 / 开始打包
打包态状态行 + 等宽日志在访达中显示 / 终止打包 / 关闭

openBuild() 会先探一次 build_status：上次关了窗口但打包还在跑时直接进打包态接着看， 不再要求用户重新配一遍。

图标：点一下换整套

const path = await invoke("pick_file", { prompt: p.name + " · 选择应用图标", exts: ["png","jpg","jpeg","heic"] });
if (!path) return;                       // 用户取消
const r = await invoke("swap_icon", { id, imagePath: path });
bdIcon = await invoke("project_icon", { id });   // 回读落盘的那张，校准预览
macIcons.delete(id); await loadMacIcons();       // 卡片头缩略图跟上

后端 icon::swap_icon 用系统自带的 sips 把任意图片居中裁方成 1024² 主图， 派生 32x32.png / 128x128.png / 128x128@2x.png，再手拼 .icns 容器与 .ico， 覆盖 src-tauri/icons/；若项目下已有构建产物（src-tauri/target/release/bundle/macos/*.app）， 顺手把 icon.icns 拷进去、codesign --force --deep --sign - 重签并 touch， Finder / Dock 换个启动就能看到新图标，不必重新打包。

应用名：写 productName，失焦即存

app_meta 读 tauri.conf.json 的 productName / identifier 填进配置态； 输入框失焦或回车触发 set_app_name 落盘（不丢改动，写失败则回滚输入框）。

后端不做 JSON 反序列化再序列化——那会把用户文件里 "build": { "frontendDist": "../ui" } 这类紧凑写法展开成多行，凭空制造一大片无关 diff。 改成只替换 productName 那一行，其余字节原样保留：

for line in json.split_inclusive('\n') {
    let trimmed = line.trim_start();
    if !trimmed.starts_with("\"productName\"") { out.push_str(line); continue; }
    // 沿用原行的缩进与尾逗号，只换值
}

只认独占一行的 productName（Tauri CLI 生成的就是这格式）。被压缩成一行或嵌套在 别的对象里时，宁可报「不是独立一行」也不猜——改坏用户的配置比不改严重得多。 名字本身还挡掉 /、:、\、" 与首字符 .：macOS 会把 : 当路径分隔符，.app 目录名里带它直接坏掉。
