/**
 * Table —— 端口雷达表格
 * 结构来自 radar.js 的 renderRadar()：4 列（端口 / 进程 / PID / 操作），
 * 端口用 <b> 加粗、PID 走 .col-pid 弱化、操作列 .col-kill 固定 90px。
 */

const ROWS = [
  { port: 1420, process: "node", pid: 55121 },
  { port: 5174, process: "node", pid: 55233 },
  { port: 3847, process: "node", pid: 55340 },
  { port: 8200, process: "node", pid: 55411 },
  { port: 5180, process: "node", pid: 55502 },
];

export function Demo() {
  return (
    <table className="table">
      <thead>
        <tr>
          <th>端口</th>
          <th>进程</th>
          <th>PID</th>
          <th className="col-kill"></th>
        </tr>
      </thead>
      <tbody>
        {ROWS.map((r) => (
          <tr key={r.port}>
            <td>
              <b>{r.port}</b>
            </td>
            <td>{r.process}</td>
            <td className="col-pid">{r.pid}</td>
            <td>
              <button className="btn is-ghost">杀掉</button>
            </td>
          </tr>
        ))}
      </tbody>
    </table>
  );
}

类名作用
.tablewidth: 100% + border-collapse: collapse
.table th--fs-11 + --fw-normal（不加深）+ --c-text-3
.table td--fs-12，8px 12px 内边距
.table tr:hover td整行 --c-surface-1 底色
.col-pidPID 列弱化到 --c-text-3（它是次要信息，端口才是主角）
.col-kill操作列固定 width: var(--px-90) = 90px

三个刻意的选择

表头不加深，反而调灰。 常规表格会把表头加粗以示区分，这里反过来——表头用 --c-text-3（32% 白） 且 font-weight: normal。因为表格整体是「数据为主、标题为辅」，表头加重会和数据抢注意力。

端口用 <b> 加粗，PID 调灰。 端口雷达的用途是快速定位「哪个端口被占了」， 所以端口值最重、进程名次之、PID 最轻。视觉权重跟着信息权重走。

操作列固定宽度。 不固定的话，按钮文字长度变化会让列宽抖动，多行之间对不齐。

空态落在表格里

$("radarBody").innerHTML =
  '<tr><td colspan="4" class="empty-state">当前没有监听中的端口</td></tr>';

注意这里把 .empty-state 直接写在 <td> 上，而不是 <td> 里再包一层 <div>。 因为 .empty-state 自带 padding: 40px 0，而 .table td 的 padding: 8px 12px 会把它压扁， 所以专门有一条覆盖：

.table td.empty-state { padding: var(--px-40) 0; }
