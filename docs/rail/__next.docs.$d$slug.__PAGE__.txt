function Btn({ item, extra = "" }: { item: Item; extra?: string }) {
  return (
    <button
      className={"rail-btn" + (item.on ? " is-on" : "") + extra}
      data-page={item.key}
      data-tip={item.label}
      aria-label={item.label}
    >
      <svg viewBox="0 0 20 20">{item.icon}</svg>
    </button>
  );
}
